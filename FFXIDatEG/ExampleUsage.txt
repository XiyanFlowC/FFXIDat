// Example of how to use Config and Localization in your main application

#include "Config.h"
#include "Localization.h"
#include "ContentView.h"

void InitializeApplication()
{
	// Load configuration
	Config& config = Config::Instance();
	if (!config.Load(L"config.ini"))
	{
		// Config file doesn't exist, create default one
		config.SetString(L"Font", L"FontName", L"Yu Gothic UI");
		config.SetInt(L"Font", L"FontSize", 16);
		config.SetString(L"UI", L"Language", L"en");
		config.Save(L"config.ini");
	}

	// Load localization strings
	Localization& loc = Localization::Instance();
	std::wstring lang = config.GetString(L"UI", L"Language", L"en");
	loc.SetLanguage(lang);
	
	std::wstring stringsFile = L"strings_" + lang + L".ini";
	if (!loc.Load(stringsFile))
	{
		// Fallback to English if requested language not found
		loc.Load(L"strings_en.ini");
	}
}

void ApplyFontSettings(ContentView& contentView)
{
	Config& config = Config::Instance();
	
	std::wstring fontName = config.GetString(L"Font", L"FontName", L"Yu Gothic UI");
	int fontSize = config.GetInt(L"Font", L"FontSize", 16);
	
	contentView.SetFont(fontName, fontSize);
}

void SaveFontSettings(const ContentView& contentView)
{
	Config& config = Config::Instance();
	
	config.SetString(L"Font", L"FontName", contentView.GetFontName());
	config.SetInt(L"Font", L"FontSize", contentView.GetFontSize());
	
	config.Save(L"config.ini");
}

// Example of using localized strings
void ShowLocalizedMessage()
{
	// LOC macro automatically gets localized string
	std::wstring okText = LOC(L"OK");
	std::wstring cancelText = LOC(L"Cancel");
	
	// Or use the Instance directly
	std::wstring errorMsg = Localization::Instance().GetString(L"Error.FileNotFound", L"File not found");
}
